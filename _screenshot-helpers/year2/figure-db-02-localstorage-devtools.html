// Step 1: Save an order to localStorage
> localStorage.setItem('lastOrder', JSON.stringify({ item: 'Propeller', qty: 2, price: 12.99 }))
← undefined
// Step 2: Read it back
> localStorage.getItem('lastOrder')
'{"item":"Propeller","qty":2,"price":12.99}'
// Step 3: Parse it back into an object
> JSON.parse(localStorage.getItem('lastOrder'))
▶ {item: 'Propeller', qty: 2, price: 12.99}
// Step 4: Check how many keys are stored
> localStorage.length
1
